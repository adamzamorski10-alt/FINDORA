## 2. SERVICE PATTERN

### 2.1 Core Responsibility

Application services are the **only** layer allowed to read legacy global arrays. They:
1. Read `window.*` global arrays (the current data source)
2. Prepare inputs for domain calculators
3. Call pure domain calculators
4. Return results to UI

Services must NOT:
- Access the DOM
- Call Firebase directly
- Persist data
- Show toasts or alerts
- Call other services (except through explicit orchestration)

### 2.2 Service Structure

```javascript
/**
 * Stage 1.4+ — <DomainThing>Service
 *
 * Bridge between legacy global application state and the pure domain calculator.
 *
 * This service reads the current in-memory financial arrays and delegates
 * the final calculation to the domain layer.
 *
 * Does NOT:
 * - manipulate the DOM
 * - render UI
 * - call Firebase directly
 * - persist data
 * - show toasts
 */

// Named export for Node.js tests
function <operation>(inputsOrCurrentState) {
  // 1. Read globals
  const data = window.<globalArray>;
  
  // 2. Prepare inputs
  const inputs = {
    ...extractFromGlobals(data)
  };
  
  // 3. Call calculator
  const calculator = window.<CalculatorName>;
  if (!calculator) {
    throw new Error('<CalculatorName> not available');
  }
  
  return calculator.compute(inputs);
}

// Optional: convenience wrapper for current state
function <operation>ForCurrentState() {
  return <operation>(window.<globalArray>);
}

// Node.js exports
if (typeof module !== 'undefined' && module.exports) {
  module.exports = { <operation>, <operation>ForCurrentState };
}

// Browser exposure
if (typeof window !== 'undefined') {
  window.<ServiceName> = { <operation>, <operation>ForCurrentState };
}
```

### 2.3 Service Naming Convention

| Pattern | Example | When to Use |
|---------|---------|-------------|
| `computeForCurrentState()` | `SafeToSpendService.computeForCurrentState()` | Single operation, no inputs needed |
| `<operation>ForCurrentState()` | `BalancesAsOfService.computeForCurrentState(asOfDate)` | Operation with optional parameters |

```javascript
    if (!budget) {
      throw new Error(`Budget not found: ${budgetId}`);

      .filter(tx => tx.type === 'expense')

      budget: budget,
      spent: spent
    });
```

```javascript
class RecurringService {
  constructor(recurringRuleRepository, transactionService) {
    this.recurringRepo = recurringRuleRepository;
    this.txService = transactionService;
  }

  async getAll() {
    return this.recurringRepo.loadAll();
  }

  async getActive() {
    return this.recurringRepo.findActive();
  }

  async isDue(rule) {
    const calculator = window.RecurringNextDateCalculator;
    if (!calculator) {
      throw new Error('RecurringNextDateCalculator not available');
    }
    const nextDate = calculator.compute({
      frequency: rule.frequency,
      dayOfMonth: rule.dayOfMonth,
      startDate: rule.lastBooked || rule.startDate
    });
    const today = new Date().toISOString().split('T')[0];
    return nextDate <= today;
  }

  async getDueThisMonth(monthKey) {
    const activeRules = await this.recurringRepo.findActive();
    const calculator = window.RecurringToMonthlyCalculator;
    if (!calculator) {
      throw new Error('RecurringToMonthlyCalculator not available');
    }
    const result = calculator.compute({
      recurringRules: activeRules,
      monthKey: monthKey
    });
    return result.total > 0 ? activeRules : [];
  }

  async book(ruleId) {
    const rule = await this.recurringRepo.findById(ruleId);
    if (!rule) {
      throw new Error(`Recurring rule not found: ${ruleId}`);
    }

    if (!(await this.isDue(rule))) {
      throw new Error(`Recurring rule is not due: ${ruleId}`);
    }

    const today = new Date().toISOString().split('T')[0];
    const transaction = await this.txService.add({
      amount: rule.amount,
      type: rule.type,
      accountId: rule.accountId,
      categoryId: rule.categoryId,
      description: rule.description,
      date: today,
      tags: ['recurring'],
      metadata: { recurringRuleId: rule.id }
    });

    await this.recurringRepo.updateLastBooked(ruleId, today);
    return transaction;
  }

  async bookAllDue() {
    const activeRules = await this.recurringRepo.findActive();
    const booked = [];
    for (const rule of activeRules) {
      try {
        const tx = await this.book(rule.id);
        booked.push(tx);
      } catch (err) {
        console.warn(`Failed to book rule ${rule.id}: ${err.message}`);
      }
    }
    return booked;
  }

  async add(inputs) {
    const rule = {
      id: generateId(),
      userId: this.userId,
      name: inputs.name,
      type: inputs.type,
      amount: inputs.amount,
      categoryId: inputs.categoryId || null,
      accountId: inputs.accountId,
      description: inputs.description || '',
      frequency: inputs.frequency,
      dayOfMonth: inputs.dayOfMonth || null,
      startDate: inputs.startDate,
      active: true,
      lastBooked: null,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    return this.recurringRepo.save(rule);
  }

  async update(id, updates) {
    const existing = await this.recurringRepo.findById(id);
    if (!existing) {
      throw new Error(`Recurring rule not found: ${id}`);
    }
    return this.recurringRepo.save({
      ...existing,
      ...updates,
      updatedAt: new Date().toISOString()
    });
  }
```

```javascript
class SavingsService {
  constructor(savingsRuleRepository, goalRepository, transactionService) {
    this.savingsRuleRepo = savingsRuleRepository;
    this.goalRepo = goalRepository;
    this.txService = transactionService;
  }

  async getAll() {
    return this.savingsRuleRepo.loadAll();
  }

  async getActive() {
    return this.savingsRuleRepo.findActive();
  }

  async getBySource(sourceId) {
    return this.savingsRuleRepo.findBySource(sourceId);
  }

  async applyRules(incomeAmount, sourceId, date) {
    const rules = await this.savingsRuleRepo.findBySource(sourceId);
    const activeRules = rules.filter(r => r.active);
    
    const calculator = window.SavingsRulesApplyCalculator;
    if (!calculator) {
      throw new Error('SavingsRulesApplyCalculator not available');
    }

    const allocations = calculator.compute({
      rules: activeRules,
      income: incomeAmount,
      sourceId: sourceId
    });

    const transactions = [];
    for (const alloc of allocations) {
      const tx = await this.txService.add({
        amount: alloc.amount,
        type: 'expense',
        accountId: sourceId,
        categoryId: null,
        description: `Auto-save: ${alloc.ruleId}`,
        date: date,
        tags: ['autosave'],
        metadata: { savingsRuleId: alloc.ruleId, destinationGoalId: alloc.destinationGoalId }
      });
      transactions.push(tx);

      // Update goal progress
      const goal = await this.goalRepo.findById(alloc.destinationGoalId);
      if (goal) {
        await this.goalRepo.updateProgress(alloc.destinationGoalId, goal.current + alloc.amount);
      }
    }

    return transactions;
  }

  async add(inputs) {
    const rule = {
      id: generateId(),
      userId: this.userId,
      sourceId: inputs.sourceId,
      percentage: inputs.percentage,
      destinationGoalId: inputs.destinationGoalId,
      label: inputs.label || '',
      active: true,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    return this.savingsRuleRepo.save(rule);
  }

  async update(id, updates) {
    const existing = await this.savingsRuleRepo.findById(id);
    if (!existing) {
      throw new Error(`Savings rule not found: ${id}`);
    }
    return this.savingsRuleRepo.save({

      ...updates,
```

```javascript
    const today = new Date().toISOString().split('T')[0];

    // Create transfer record
    const transfer = {
      id: transferId,
      userId: this.userId,
      fromAccountId: inputs.fromAccountId,
      toAccountId: inputs.toAccountId,
      amount: inputs.amount,
      description: inputs.description || '',
      date: today,
      createdAt: today
    };

    // Create paired transactions
    const fromTx = await this.txService.add({
      amount: inputs.amount,
      type: 'expense',
      accountId: inputs.fromAccountId,
      description: `Transfer: ${inputs.description}`,
      date: today,
      tags: ['transfer'],
      metadata: { transferId: transferId }
    });

    const toTx = await this.txService.add({
      amount: inputs.amount,
      type: 'income',
      accountId: inputs.toAccountId,
      description: `Transfer: ${inputs.description}`,
      date: today,
      tags: ['transfer'],
      metadata: { transferId: transferId }
    });

    await this.transferRepo.save(transfer);

    return { transfer, fromTx, toTx };
  }

  async remove(id) {
    return this.transferRepo.remove(id);
  }
}
```

```javascript
class ReportingService {
  constructor(transactionRepository, accountRepository) {
    this.txRepo = transactionRepository;
    this.accRepo = accountRepository;
  }

  async getMonthlySummary(monthKey) {
    const transactions = await this.txRepo.loadAll();
    const monthTx = window.TransactionFilterByMonth
      ? window.TransactionFilterByMonth.compute({ transactions, monthKey })
      : transactions.filter(t => t.date && t.date.startsWith(monthKey) && !t.metadata?.excluded);

    const income = monthTx
      .filter(t => t.type === 'income')
      .reduce((s, t) => s + t.amount, 0);
    const expense = monthTx
      .filter(t => t.type === 'expense')
      .reduce((s, t) => s + t.amount, 0);

    const breakdownCalculator = window.MonthCategoryBreakdownCalculator;
    const byCategory = breakdownCalculator
      ? breakdownCalculator.compute({ transactions: monthTx, monthKey })
      : { expenseByCategory: {}, incomeByCategory: {} };

    return {
      monthKey,
      income,
      expense,
      net: income - expense,
      byCategory
    };
  }

  async getWealthTrend(monthKeys) {
    const transactions = await this.txRepo.loadAll();
    const calculator = window.WealthMonthTransactionStatsCalculator;
    if (!calculator) {
      throw new Error('WealthMonthTransactionStatsCalculator not available');
    }

    return monthKeys.map(mk => {
      const result = calculator.compute({ transactions, monthKey: mk });
      return { monthKey: mk, ...result };
    });
  }

  async getCategoryTrend(categoryId, monthKeys) {
    const transactions = await this.txRepo.loadAll();
    const trend = [];
    for (const mk of monthKeys) {
      const monthTx = window.TransactionFilterByMonth
        ? window.TransactionFilterByMonth.compute({ transactions, monthKey: mk })
        : transactions.filter(t => t.date && t.date.startsWith(mk) && !t.metadata?.excluded);
      
      const amount = monthTx
        .filter(t => t.categoryId === categoryId)
        .reduce((s, t) => s + (t.type === 'expense' ? -t.amount : t.amount), 0);
      
      trend.push({ monthKey: mk, amount });
    }
    return trend;
  }
}
```

## 5. SERVICE FACTORY (Composition Root)

Services need a factory to wire dependencies:

```javascript
// src/application/service-factory.js

class ServiceFactory {
  constructor(storageAdapter, userId) {
    this.storageAdapter = storageAdapter;
    this.userId = userId;
    this._repos = {};
    this._services = {};
  }

  _getRepo(name) {
    if (!this._repos[name]) {
      switch (name) {
        case 'transaction':
          this._repos[name] = new TransactionRepository(this.storageAdapter, this.userId);
          break;
        case 'account':
          this._repos[name] = new AccountRepository(this.storageAdapter, this.userId);
          break;
        case 'budget':
          this._repos[name] = new BudgetRepository(this.storageAdapter, this.userId);
          break;
        case 'goal':
          this._repos[name] = new GoalRepository(this.storageAdapter, this.userId);
          break;
        case 'recurringRule':
          this._repos[name] = new RecurringRuleRepository(this.storageAdapter, this.userId);
          break;
        case 'transfer':
          this._repos[name] = new TransferRepository(this.storageAdapter, this.userId);
          break;
        case 'savingsRule':
          this._repos[name] = new SavingsRuleRepository(this.storageAdapter, this.userId);
          break;
        case 'person':
          this._repos[name] = new PersonRepository(this.storageAdapter, this.userId);
          break;
        default:
          throw new Error(`Unknown repository: ${name}`);
      }
    }
    return this._repos[name];
  }

  getTransactionService() {
    if (!this._services.transaction) {
      this._services.transaction = new TransactionService(this._getRepo('transaction'));
    }
    return this._services.transaction;
  }

  getBudgetService() {
    if (!this._services.budget) {
      this._services.budget = new BudgetService(
        this._getRepo('budget'),
        this._getRepo('transaction')
      );
    }
    return this._services.budget;
  }

  getGoalService() {
    if (!this._services.goal) {
      this._services.goal = new GoalService(this._getRepo('goal'));
    }
    return this._services.goal;
  }

  getRecurringService() {
    if (!this._services.recurring) {
      this._services.recurring = new RecurringService(
        this._getRepo('recurringRule'),
        this.getTransactionService()
      );
    }
    return this._services.recurring;
  }

  getSavingsService() {
    if (!this._services.savings) {
      this._services.savings = new SavingsService(
        this._getRepo('savingsRule'),
        this._getRepo('goal'),
        this.getTransactionService()
      );
    }
    return this._services.savings;
  }

  getDebtService() {
    if (!this._services.debt) {
      this._services.debt = new DebtService(this._getRepo('person'));
    }
    return this._services.debt;
  }

  getTransferService() {
    if (!this._services.transfer) {
      this._services.transfer = new TransferService(
        this._getRepo('transfer'),
        this.getTransactionService(),
        this._getRepo('account')
      );
    }
    return this._services.transfer;
  }

  getReportingService() {
    if (!this._services.reporting) {
      this._services.reporting = new ReportingService(
        this._getRepo('transaction'),
        this._getRepo('account')
      );
    }
    return this._services.reporting;
  }
}
```
