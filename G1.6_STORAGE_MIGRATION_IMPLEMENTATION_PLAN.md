## 3. INDEXEDDB STORAGE ADAPTER

### 3.1 Design

```javascript
/**
 * Stage 1.6 — IndexedDBStorageAdapter
 *
 * Local-first persistence using IndexedDB.
 * Implements StorageAdapter contract.
 * Uses `idb` library for promise-based IndexedDB access.
 */

class IndexedDBStorageAdapter {
  constructor(dbName = 'finora', storeName = 'data') {
    this.dbName = dbName;
    this.storeName = storeName;
    this.db = null;
  }

  async init() {
    this.db = await idb.openDB(this.dbName, 1, {
      upgrade(db) {
        if (!db.objectStoreNames.contains(this.storeName)) {
          db.createObjectStore(this.storeName);
        }
      }
    });
  }

  async get(key) {
    const tx = this.db.transaction(this.storeName, 'readonly');
    const store = tx.objectStore(this.storeName);
    return store.get(key);
  }

  async set(key, value) {
    const tx = this.db.transaction(this.storeName, 'readwrite');
    const store = tx.objectStore(this.storeName);
    await store.put(value, key);
    await tx.done;
  }

  async update(key, updater) {
    const existing = await this.get(key);
    const updated = typeof updater === 'function' ? updater(existing) : { ...existing, ...updater };
    await this.set(key, updated);
  }

  async updateMany(updates) {
    const tx = this.db.transaction(this.storeName, 'readwrite');
    const store = tx.objectStore(this.storeName);
    for (const [key, updater] of Object.entries(updates)) {
      const existing = await store.get(key);
      const updated = typeof updater === 'function' ? updater(existing) : { ...existing, ...updater };
      await store.put(updated, key);
    }
    await tx.done;
  }

  async remove(key) {
    const tx = this.db.transaction(this.storeName, 'readwrite');
    const store = tx.objectStore(this.storeName);
    await store.delete(key);
    await tx.done;
  }

  listen(path, callback, errCallback) {
    // IndexedDB doesn't natively support listeners.
    // Use BroadcastChannel or storage event for cross-tab sync.
    const bc = new BroadcastChannel('finora-sync');
    bc.onmessage = (event) => {
      if (event.data.path === path) {
        this.get(path).then(callback).catch(errCallback);
      }
    };

    // Initial read
    this.get(path).then(callback).catch(errCallback);

    return {
      unsubscribe: () => bc.close()
    };
  }

  listenRoot(callback, errCallback) {
    // Listen to all keys by querying all
    const bc = new BroadcastChannel('finora-sync');
    bc.onmessage = async () => {
      try {
        const all = await this.getAll();
        callback(all);
      } catch (err) {
        if (errCallback) errCallback(err);
      }
    };

    this.getAll().then(callback).catch(errCallback);

    return {
      unsubscribe: () => bc.close()
    };
  }

  async getAll() {
    const tx = this.db.transaction(this.storeName, 'readonly');
    const store = tx.objectStore(this.storeName);
    const all = await store.getAll();
    const result = {};
    for (const entry of all) {
      result[entry.key] = entry.value;
    }
    return result;
  }
}
```

    await adapter.set('key1', { value: 1 });
    await adapter.set('key1', { value: 2 });
    assert.strictEqual(updates.length, 2);
    unsubscribe();

function migratePeopleV1toV2(data) {
  const debtors = Array.isArray(data.debtors) ? data.debtors : [];
  const creditors = Array.isArray(data.creditors) ? data.creditors : [];

  const people = [
    ...debtors.map(d => ({
      ...d,
      type: 'debtor',
      balances: {
        total: d.debts ? d.debts.reduce((s, b) => s + b.amount, 0) : 0,
        paid: d.debts ? d.debts.reduce((s, b) => s + b.repaid, 0) : 0,
        remaining: 0 // computed by calculator
      },
      debts: d.debts || [],
      repayments: d.repayments || []
    })),
    ...creditors.map(c => ({
      ...c,
      type: 'creditor',
      balances: {
        total: c.loans ? c.loans.reduce((s, l) => s + l.amount, 0) : 0,
        paid: c.loans ? c.loans.reduce((s, l) => s + l.paid, 0) : 0,
        remaining: 0 // computed by calculator
      },
      debts: c.loans || [],
      repayments: c.repayments || []
    }))
  ];

  return people;
}

module.exports = {
  fromVersion: '1.0.0',
  toVersion: '2.0.0',

## 6. MIGRATION FLOW

### 6.1 On First Load

```javascript
async function migrateUserData(storageAdapter, userId) {
  const migrator = new Migrator(registry);
  const CURRENT_VERSION = '2.0.0';

  // Migrate each key
  const keysToMigrate = [
    'finapp_transactions',
    'finapp_dashboard_layouts',
    'finapp_autosave_rules',
    'finapp_debtors',
    'finapp_creditors',
    'finapp_accounts',
    'finapp_categories'
  ];

  const updates = {};
  for (const key of keysToMigrate) {
    const data = await storageAdapter.get(key);
    if (!data) continue;

    const version = getDataVersion(data);
    if (version !== CURRENT_VERSION) {
      const result = migrator.migrate(data, version, CURRENT_VERSION, key);
      if (result.errors.length > 0) {
        console.error(`Migration failed for ${key}:`, result.errors);
        continue;
      }
      updates[key] = result.data;
    }
  }

  if (Object.keys(updates).length > 0) {
    await storageAdapter.updateMany(updates);
  }
}
```

### 6.2 Atomic Migration

- Migrations are pure functions.
- Source data is never mutated (deep clone before transform).
- Each key is migrated independently.
- Failed migrations preserve original data.
- Successfully migrated keys are written atomically via `updateMany`.

### 7.3 Hybrid Mode

When using Firebase as sync layer:
- IndexedDB is local cache.
- Firebase is source of truth.
- On app load: read from IndexedDB first (instant), then sync from Firebase.
- On write: write to both (local first, then Firebase).

```javascript
class HybridStorageAdapter {
  constructor(localAdapter, remoteAdapter) {
    this.local = localAdapter;
    this.remote = remoteAdapter;
  }

  async get(key) {
    // Prefer local cache
    const local = await this.local.get(key);
    if (local) return local;
    // Fallback to remote
    const remote = await this.remote.get(key);
    if (remote) {
      await this.local.set(key, remote);
    }
    return remote;
  }

  async set(key, value) {
    await this.local.set(key, value);
    await this.remote.set(key, value);
  }
}
```
